cmake_minimum_required(VERSION 3.20)
project(test-rlimit-as C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-rlimit-as src/main.c)
target_include_directories(test-rlimit-as PRIVATE src)
target_compile_options(test-rlimit-as PRIVATE -Wall -Wextra)
install(TARGETS test-rlimit-as RUNTIME DESTINATION usr/bin/starry-test-suit)
