cmake_minimum_required(VERSION 3.20)
project(test-nonblock-connect C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-nonblock-connect src/main.c)
target_include_directories(test-nonblock-connect PRIVATE src)
target_compile_options(test-nonblock-connect PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-nonblock-connect RUNTIME DESTINATION usr/bin/starry-test-suit)
