cmake_minimum_required(VERSION 3.20)
project(test-sysfs-node-meminfo C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-sysfs-node-meminfo src/main.c)
target_include_directories(test-sysfs-node-meminfo PRIVATE ../common)
target_compile_options(test-sysfs-node-meminfo PRIVATE -Wall -Wextra -Werror -pthread)
target_link_options(test-sysfs-node-meminfo PRIVATE -pthread)
install(TARGETS test-sysfs-node-meminfo RUNTIME DESTINATION usr/bin/starry-test-suit)
