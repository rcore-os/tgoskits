cmake_minimum_required(VERSION 3.20)
project(dynamic-debug-control C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(dynamic-debug-control src/dynamic_debug_control.c)
target_compile_options(dynamic-debug-control PRIVATE -Wall -Wextra -Werror)
target_link_libraries(dynamic-debug-control PRIVATE pthread)
install(TARGETS dynamic-debug-control RUNTIME DESTINATION usr/bin/starry-test-suit)
