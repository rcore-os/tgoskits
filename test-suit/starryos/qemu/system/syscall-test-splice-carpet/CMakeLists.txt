cmake_minimum_required(VERSION 3.20)
project(test-splice-carpet C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-splice-carpet src/main.c)
target_include_directories(test-splice-carpet PRIVATE src)
target_compile_options(test-splice-carpet PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-splice-carpet RUNTIME DESTINATION usr/bin/starry-test-suit)
