cmake_minimum_required(VERSION 3.20)
project(syscall-test-mlock-family C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(syscall-test-mlock-family src/main.c)
target_include_directories(syscall-test-mlock-family PRIVATE src)
target_compile_options(syscall-test-mlock-family PRIVATE -Wall -Wextra -Werror)
install(TARGETS syscall-test-mlock-family RUNTIME DESTINATION usr/bin/starry-test-suit)
