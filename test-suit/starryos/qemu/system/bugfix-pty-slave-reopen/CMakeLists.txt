cmake_minimum_required(VERSION 3.20)
project(bugfix-pty-slave-reopen C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(bugfix-pty-slave-reopen src/main.c)
target_include_directories(bugfix-pty-slave-reopen PRIVATE src ${CMAKE_CURRENT_SOURCE_DIR}/../common)
target_compile_options(bugfix-pty-slave-reopen PRIVATE -Wall -Wextra -Werror)
install(TARGETS bugfix-pty-slave-reopen RUNTIME DESTINATION usr/bin/starry-test-suit)
