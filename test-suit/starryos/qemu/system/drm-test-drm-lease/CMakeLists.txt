cmake_minimum_required(VERSION 3.20)
project(test-drm-lease C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-drm-lease src/main.c)
target_include_directories(test-drm-lease PRIVATE src)
target_compile_options(test-drm-lease PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-drm-lease RUNTIME DESTINATION usr/bin/starry-test-suit)
