cmake_minimum_required(VERSION 3.20)
project(test-rdev-nvme C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-rdev-nvme src/main.c)
target_include_directories(test-rdev-nvme PRIVATE src)
target_compile_options(test-rdev-nvme PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-rdev-nvme RUNTIME DESTINATION usr/bin/starry-test-suit)
