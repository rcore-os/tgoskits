cmake_minimum_required(VERSION 3.20)
project(test-clone-tls C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(test-clone-tls src/main.c)
target_include_directories(test-clone-tls PRIVATE src)
target_compile_options(test-clone-tls PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test-clone-tls PRIVATE Threads::Threads)
install(TARGETS test-clone-tls RUNTIME DESTINATION usr/bin/starry-test-suit)
