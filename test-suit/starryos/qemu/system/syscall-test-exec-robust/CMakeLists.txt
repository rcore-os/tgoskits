cmake_minimum_required(VERSION 3.20)
project(syscall-test-exec-robust C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(syscall-test-exec-robust src/main.c)
target_compile_options(syscall-test-exec-robust PRIVATE -Wall -Wextra -Werror)
target_link_libraries(syscall-test-exec-robust PRIVATE pthread)
install(TARGETS syscall-test-exec-robust RUNTIME DESTINATION usr/bin/starry-test-suit)
