cmake_minimum_required(VERSION 3.20)
project(test-mprotect-wx C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-mprotect-wx src/main.c)
target_include_directories(test-mprotect-wx PRIVATE src)
target_compile_options(test-mprotect-wx PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-mprotect-wx RUNTIME DESTINATION usr/bin/starry-test-suit)
