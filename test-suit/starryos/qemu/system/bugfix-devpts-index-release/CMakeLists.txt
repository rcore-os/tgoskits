cmake_minimum_required(VERSION 3.20)
project(bugfix-devpts-index-release C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(bugfix-devpts-index-release src/main.c)
target_include_directories(bugfix-devpts-index-release PRIVATE src ${CMAKE_CURRENT_SOURCE_DIR}/../common)
target_compile_options(bugfix-devpts-index-release PRIVATE -Wall -Wextra -Werror)
install(TARGETS bugfix-devpts-index-release RUNTIME DESTINATION usr/bin/starry-test-suit)
