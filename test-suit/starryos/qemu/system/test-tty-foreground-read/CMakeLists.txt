cmake_minimum_required(VERSION 3.20)
project(test-tty-foreground-read C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-tty-foreground-read src/main.c)
target_include_directories(test-tty-foreground-read PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common)
target_compile_options(test-tty-foreground-read PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test-tty-foreground-read PRIVATE pthread)
install(TARGETS test-tty-foreground-read RUNTIME DESTINATION usr/bin/starry-test-suit)
