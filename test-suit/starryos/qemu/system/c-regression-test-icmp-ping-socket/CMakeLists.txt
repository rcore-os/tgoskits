cmake_minimum_required(VERSION 3.20)
project(test-icmp-ping-socket C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-icmp-ping-socket src/main.c)
target_include_directories(test-icmp-ping-socket PRIVATE src)
target_compile_options(test-icmp-ping-socket PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-icmp-ping-socket RUNTIME DESTINATION usr/bin/starry-test-suit)
