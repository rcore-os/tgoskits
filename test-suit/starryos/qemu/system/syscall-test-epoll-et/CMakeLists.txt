cmake_minimum_required(VERSION 3.20)
project(test-epoll-et C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-epoll-et src/main.c)
target_include_directories(test-epoll-et PRIVATE src)
target_compile_options(test-epoll-et PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-epoll-et RUNTIME DESTINATION usr/bin/starry-test-suit)
