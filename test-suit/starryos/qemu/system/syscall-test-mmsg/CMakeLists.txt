cmake_minimum_required(VERSION 3.20)
project(test-mmsg C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-mmsg src/main.c)
target_include_directories(test-mmsg PRIVATE src)
target_compile_options(test-mmsg PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-mmsg RUNTIME DESTINATION usr/bin/starry-test-suit)
