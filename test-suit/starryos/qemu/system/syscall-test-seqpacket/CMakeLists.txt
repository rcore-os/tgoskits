cmake_minimum_required(VERSION 3.20)
project(test-seqpacket C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-seqpacket src/main.c)
target_include_directories(test-seqpacket PRIVATE src)
target_compile_options(test-seqpacket PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-seqpacket RUNTIME DESTINATION usr/bin/starry-test-suit)
