cmake_minimum_required(VERSION 3.20)
project(starry_vdec C)

set(fixtures "${CMAKE_CURRENT_BINARY_DIR}/fixtures.h")
file(WRITE "${fixtures}" "/* Embedded baseline JPEG fixtures. */\n")
foreach(format yuv420 yuv422h yuv422v yuv444)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/${format}.jpg" bytes HEX)
    string(REGEX REPLACE "(..)" "0x\\1," bytes "${bytes}")
    file(APPEND "${fixtures}" "static const unsigned char ${format}[] = {${bytes}};\n")
endforeach()
add_executable(starry-vdec main.c)
target_include_directories(starry-vdec PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(starry-vdec PRIVATE -Wall -Wextra -Werror)
target_link_options(starry-vdec PRIVATE -static)
install(TARGETS starry-vdec RUNTIME DESTINATION bin)
